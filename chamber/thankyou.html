"use strict"; // ========================================= // READ SUBMITTED FORM DATA // ========================================= const formParameters = new URLSearchParams(window.location.search); // Map the form's name attributes to the confirmation page's element IDs. const confirmationFields = { "first-name": "firstName", "last-name": "lastName", email: "email", phone: "phone", organization: "organization", timestamp: "timestamp" }; // Safely display submitted values. Object.entries(confirmationFields).forEach(([elementId, parameterName]) => { const outputElement = document.getElementById(elementId); const submittedValue = formParameters.get(parameterName); if (!outputElement) { return; } if (!submittedValue) { outputElement.textContent = "Not provided"; return; } // Format the timestamp for readability. if (parameterName === "timestamp") { const submittedDate = new Date(submittedValue); if (!Number.isNaN(submittedDate.getTime())) { outputElement.textContent = submittedDate.toLocaleString( undefined, { dateStyle: "medium", timeStyle: "short" } ); } else { outputElement.textContent = submittedValue; } return; } // textContent prevents submitted text from being interpreted as HTML. outputElement.textContent = submittedValue; }); // ========================================= // FOOTER DATES // ========================================= const currentYear = document.querySelector("#current-year"); const lastModified = document.querySelector("#last-modified"); if (currentYear) { currentYear.textContent = new Date().getFullYear(); } if (lastModified) { lastModified.textContent = document.lastModified; }
